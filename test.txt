#include <stdio.h>
#include <float.h>
int main() {
   char op;
   double a, b, result;
   // Input operator and operands
   printf("Enter an operator (+, -, *, /): ");
   scanf(" %c", &op);
   printf("Enter two operands: ");
   scanf("%lf %lf", &a, &b);
   // Perform operation based on the operator
   switch (op) {
       case '+':
           result = a + b;
           break;
       case '-':
           result = a - b;
           break;
       case '*':
           result = a * b;
           break;
       case '/':
           if (b != 0)
               result = a / b;
           else {
               printf("Error! Division by zero.\n");
               return 1;
           }
           break;
       default:
           printf("Error! Invalid operator.\n");
           return 1;
   }
   // Display the result
   printf("Result: %.2lf\n", result);
   return 0;
}
